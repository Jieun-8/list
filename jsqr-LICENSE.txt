  paint(root) {
    if (!root) return;
    const shuffled = [...window.CARD_VISIBILITY.mask(this.cards)];
    for (let i = shuffled.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
    }
    const groups = root.querySelectorAll(".marquee-group");
    if (groups.length !== 2) return;
    root.addEventListener(
      "load",
      (event) => {
        const img = event.target;
        if (!img.matches?.(".marquee-card img")) return;
        const card = window.CARD_VISIBILITY.mask(this.cards).find(
            (c) => c.id === img.closest(".marquee-card").dataset.preview,
          ),
          frame = img.closest(".marquee-image-frame");
        const horizontal =
          card?.game === "aikatsu" && img.naturalWidth > img.naturalHeight;
        frame.classList.toggle("landscape", horizontal);
        const width = horizontal ? img.naturalHeight : img.naturalWidth,
          height = horizontal ? img.naturalWidth : img.naturalHeight;
        if (width && height) {
          frame.style.aspectRatio = width + " / " + height;
          frame.style.setProperty("--image-ratio", String(height / width));
        }
      },
      true,
    );
    const buttons = shuffled.map((card) => {
      const button = document.createElement("button");
      button.className = "marquee-card";
      button.type = "button";
      button.dataset.preview = card.id;
      button.dataset.game = card.game;
      button.setAttribute("aria-label", (card.name || card.code) + "の詳細");
      const frame = document.createElement("span");
      frame.className = "marquee-image-frame";
      const img = document.createElement("img");
      img.alt = card.name || card.code;
      img.decoding = "async";
      img.loading = "eager";
      img.addEventListener("load", () => {
        const horizontal =
          card.game === "aikatsu" && img.naturalWidth > img.naturalHeight;
        frame.classList.toggle("landscape", horizontal);
        const width = horizontal ? img.naturalHeight : img.naturalWidth,
          height = horizontal ? img.naturalWidth : img.naturalHeight;
        if (width && height) {
          frame.style.aspectRatio = width + " / " + height;
          frame.style.setProperty("--image-ratio", String(height / width));
        }
      });
      img.src = card.front;
      frame.append(img);
      button.append(frame);
      return button;
    });
    groups[0].replaceChildren(...buttons);
    groups[1].replaceChildren(
      ...buttons.map((button) => {
        const clone = button.cloneNode(true);
        clone.tabIndex = -1;
        return clone;
      }),
    );
  },
});
